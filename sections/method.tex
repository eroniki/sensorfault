\subsection{Setting, Measurement Model and Faults}
\label{sec:method:setting}
A static scene is observed by $N$ posed RGB images $I_i$ with known intrinsics, in the main setting aerial
survey images flown in parallel strips. A monocular metric depth model gives one depth map $\hat d_i$ per
image. The surface is a set of 2DGS surfels \cite{gs_2dgs} fitted by depth-supervised splatting and
rendered with gsplat \cite{gs_gsplat} into expected depth, alpha weights and accumulated opacity. Colour
is a separate appearance block: the photometric loss updates the colours and may refine the poses, but its
gradients are stopped at the surfel geometry and opacity, so reflectance cannot corrupt the geometry. The
gate uses depth only. The nominal measurement of pixel $u$ in view $i$ is
\begin{equation}
\begin{split}
  \hat d_i(u)&=a_i\bigl(d_i(u)+\beta\,d_i(u)^{2}\bigr)+b_i+\varepsilon_i(u),\\
  \boldsymbol\varepsilon_i&\sim\mathcal{N}\bigl(\mathbf{0},\ \gamma^{2}DCD\bigr),
\end{split}
  \label{eq:meas}
\end{equation}
with a per-view scale and offset $(a_i,b_i)$ absorbed by alignment, a \emph{shared distance bias} $\beta$,
and a noise covariance whose shape is calibrated offline:
$D=\mathrm{diag}\,\sigma(\text{range},\text{texture},\text{incidence})$, and
$C=(1-\nu_{\mathrm{nug}})K_\ell+\nu_{\mathrm{nug}}I$ is a spatial correlation with length scale $\ell$ and
nugget $\nu_{\mathrm{nug}}$. The only free noise variable is the scene temperature $\gamma$
\cite{cal_guo_temperature}; there are no per-view noise parameters, because any parameter fitted to a
view's own residual would absorb that view's fault. A \emph{fault} is a regional deviation that
\eqref{eq:meas} cannot explain: a connected region of a view whose depth is wrong by many $\sigma$,
typically on a specular or textureless surface. A whole-view affine error is absorbed by $(a_i,b_i)$ and
is not a fault.

\subsection{Poses}
\label{sec:method:poses}
Poses are refined, not estimated from scratch. Each strip's trajectory is a uniform cubic B-spline in time
\cite{traj_furgale_basis}, cumulative on $SO(3)$ \cite{traj_sommer_bspline}, fitted to the GNSS/IMU poses
and refined in a differentiable pose graph with the GNSS/IMU prior, a smoothness prior and
render-and-compare depth factors; no feature correspondences are used. When view $j$ is tested, its own
render-and-compare factor is removed, so its pose cannot absorb its depth fault, and the remaining pose
uncertainty $\Sigma_q=\mathbf{b}(t_j)\,\Sigma_{\mathrm{ctrl}}\,\mathbf{b}(t_j)^{\top}$, with
$\mathbf{b}(t_j)$ the spline's basis row at the view's time stamp, enters the test
(Supplement~\ref{S-sup:method}).

\subsection{Surfels with an Uncertainty That Renders}
\label{sec:method:surfels}
Surfel $k$ has a \emph{shape} used only for rendering: centre $\boldsymbol\mu_k$, rotation
$\mathbf{R}_k=[\mathbf{t}_1\ \mathbf{t}_2\ \mathbf{n}]$ with normal $\mathbf{n}$, scales $\mathbf{s}_k$ and
opacity $\alpha_k$. Separately, it carries an \emph{uncertainty} over the perturbation
\begin{equation}
\begin{split}
  \boldsymbol\theta_k&=\bigl(\delta\boldsymbol\mu,\ \delta\boldsymbol\phi,\ \delta\log\mathbf{s},\ \delta\,\mathrm{logit}\,\alpha\bigr)\in\R^{10},\\
  \mathbf{R}_k&\leftarrow\mathbf{R}_k\exp([\delta\boldsymbol\phi]_\times),
\end{split}
\end{equation}
kept in information form as a $10\times10$ block $\Lambda_k$. Opacity is trained during a warm-up and then
frozen (Algorithm~\ref{alg:gate}); it is not re-estimated when a view is left out.

Depth observes the normal offset and the tilt of a surfel. With $\nu=\mathbf{n}^{\top}\delta\boldsymbol\mu$
and the tangent components of the normal perturbation, $\tau_1=\delta\phi_2$ and $\tau_2=-\delta\phi_1$,
the sub-block is $(\nu,\tau_1,\tau_2)=G_k\boldsymbol\theta_k$ with a fixed $3\times10$ map $G_k$. The ray of
pixel $u$ of view $i$, $\mathbf{w}=\mathbf{R}_i\mathbf{K}^{-1}[u\ v\ 1]^{\top}$, meets the surfel plane at
the z-depth $d_k=\mathbf{n}^{\top}(\boldsymbol\mu_k-\mathbf{c}_i)/\mathbf{n}^{\top}\mathbf{w}$ and the point
$\mathbf{x}$, and to first order
\begin{equation}
\begin{split}
  \frac{\partial d_k}{\partial(\nu,\tau_1,\tau_2)}&=
  \frac{\bigl[\,1,\ -\mathbf{t}_1^{\top}(\mathbf{x}-\boldsymbol\mu_k),\ -\mathbf{t}_2^{\top}(\mathbf{x}-\boldsymbol\mu_k)\bigr]}{\mathbf{n}^{\top}\mathbf{w}},\\
  J_{uk}&=\frac{\partial d_k}{\partial(\nu,\tau_1,\tau_2)}\,G_k .
\end{split}
  \label{eq:jac}
\end{equation}
Opacity acts through the compositing weights, whose derivative
$\partial\hat D_i(u)/\partial\,\mathrm{logit}\,\alpha_k$ comes from the rasteriser's backward pass; the
scale columns vanish, so the scales keep a weak prior (Supplement~\ref{S-sup:derivations}). The rasteriser
composites $\hat D_i(u)=\sum_k w_{uk}d_k$ with normalised alpha weights $w_{uk}$; with opacity frozen, the
rendered depth variance is
\begin{equation}
  V_i(u)=\sum_k w_{uk}^{2}\,J_{uk}\,P_k\,J_{uk}^{\top},
  \label{eq:var}
\end{equation}
with $P_k$ the inverse of $\Lambda_k$ restricted to the nine geometric coordinates. An offset moves every
pixel's depth alike and a tilt moves it in proportion to the lever arm, so the rendered variance is
smallest at the centre of a surfel and grows towards its rim.

\subsection{Fusion and Leave-Out by Subtraction}
\label{sec:method:fusion}
At the fitted surfels (the MAP), view $i$ adds to surfel $k$ the Laplace information
\begin{equation}
\begin{split}
  \Lambda_{k}^{(i)}&=\sum_{u}\frac{a_i^{2}\,\mathbf{g}_{uk}^{\top}\mathbf{g}_{uk}}{\gamma^{2}\sigma_{i}^{2}(u)},\qquad
  \boldsymbol\eta_{k}^{(i)}=\sum_{u}\frac{a_i\,\mathbf{g}_{uk}^{\top}e_i(u)}{\gamma^{2}\sigma_{i}^{2}(u)},\\
  \mathbf{g}_{uk}&=\Bigl[w_{uk}J_{uk},\ \frac{\partial\hat D_i(u)}{\partial\,\mathrm{logit}\,\alpha_k}\Bigr],
\end{split}
  \label{eq:info}
\end{equation}
with $e_i(u)$ the residual of view $i$ at the MAP, so that $\boldsymbol\eta_k^{(i)}=-\nabla_k\mathcal{L}_i$,
and $\Lambda_k=\Lambda_0+\sum_i\Lambda_k^{(i)}$. This is the alpha-weighted Gauss--Newton information of the
rasteriser; it needs no correspondences, because each pixel reaches the surfels it renders. Leaving view
$j$ out is a subtraction and one Newton step on the mean,
\begin{equation}
  \Lambda_{k}^{(-j)}=\Lambda_k-\Lambda_k^{(j)},\qquad
  \boldsymbol\mu_{k}^{(-j)}=\boldsymbol\mu_k+\bigl(\Lambda_{k}^{(-j)}\bigr)^{-1}\nabla_{k}\mathcal{L}_j(\boldsymbol\mu),
  \label{eq:loo}
\end{equation}
on the nine geometric coordinates, with $\mathcal{L}_j$ the depth loss of view $j$. Both terms come from one
backward pass of view $j$. The step is the infinitesimal jackknife, exact for the linear Gaussian model
(Supplement~\ref{S-sup:derivations}), and the resulting statistic equals the solution-separation statistic
\cite{flt_raim_joerger}. In message-passing terms, each view sends the surfels an information-form message
and the subtraction is the cavity step of expectation propagation \cite{mp_minka_ep}, taken once at the
MAP without moment matching (Supplement~\ref{S-sup:method}).

\runin{Leave-block-out.} Neighbouring frames see the same ground from almost the same direction, so when
the depth model fails on a surface it tends to fail in all of them alike (H2). Left out alone, a faulty view
is then tested against a cavity that its faulty neighbours still support, and the fault is masked. We
therefore leave out a block $\mathcal{B}_j$ of $B$ neighbouring frames with view $j$, on aerial data a
stretch of the same strip centred on $j$:
\begin{equation}
  \begin{aligned}
  \Lambda_k^{(-\mathcal{B}_j)}&=\Lambda_k-\textstyle\sum_{i\in\mathcal{B}_j}\Lambda_k^{(i)},\\
  \boldsymbol\mu_k^{(-\mathcal{B}_j)}&=\boldsymbol\mu_k+\bigl(\Lambda_k^{(-\mathcal{B}_j)}\bigr)^{-1}\textstyle\sum_{i\in\mathcal{B}_j}\nabla_k\mathcal{L}_i ,
  \end{aligned}
  \label{eq:block}
\end{equation}
again exact for the linear Gaussian model. Only view $j$ is tested against this cavity, so the number of
tests does not depend on $B$. The price is a weaker cavity: the rendered variance and the minimum
detectable bias rise, and pixels seen only within the block become untestable. $B$ is chosen on
development data from $B\in\{1,3,5,9\}$; $B=1$ is the ablation. Fig.~\ref{fig:cavity} follows one strip
through fusion, cavity and test.

\subsection{The View Gate}
\label{sec:method:gate}
To test view $j$, its cavity, written $(-j)$, is rendered into it. A surfel enters this render only if
(i)~the views outside the block give it normal-offset information,
$\mathbf{e}_1^{\top}G_k(\Lambda_k^{(-j)}-\Lambda_0)G_k^{\top}\mathbf{e}_1\ge\lambda_{\min}$, and (ii)~the block
supplied at most a fraction $\rho=\tfrac12$ of its opacity information,
$\Lambda_k^{(\mathcal{B}_j)}[\alpha,\alpha]/\Lambda_k[\alpha,\alpha]\le\rho$. Surfels that fail either rule were
made or shaped by the block; they are dropped and the compositing weights recomputed, so the tested view
cannot hide its fault behind surfels it created. A pixel whose accumulated opacity then falls below
$\tau_W$ is \emph{untestable}: no other view sees it. On the testable pixels, with the cavity depth
$\hat D_{-j}(u)$ and variance $V_{-j}(u)$ from \eqref{eq:var}, the alignment $(a_j,b_j)$ is fitted by
weighted least squares and the residual
\begin{equation}
\begin{split}
  r_j(u)&=\hat d_j(u)-\bigl(a_j\hat D_{-j}(u)+b_j\bigr),\\
  \Sigma_r&=\hat\gamma_{-j}^{2}DCD+a_j^{2}\,U P^{(-j)} U^{\top}+J_q\Sigma_qJ_q^{\top}
\end{split}
  \label{eq:res}
\end{equation}
is whitened. Here $U$ is the block-sparse pixel-to-surfel Jacobian, whose row for pixel $u$ holds
$w_{uk}J_{uk}$ in the columns of surfel $k$, $P^{(-j)}$ is block-diagonal with the cavity covariances
$P_k^{(-j)}$, and $J_q$ is the Jacobian of the rendered depth with respect to the pose. Pixels rendered by
the same surfel share its estimation error, so the off-diagonal terms of $UP^{(-j)}U^{\top}$ matter; the
Woodbury identity keeps whitening linear in the number of pixels (Supplement~\ref{S-sup:method}). The scene
temperature is fitted without view $j$: $\hat\gamma_{-j}^{2}$ solves
$\mathrm{median}_{i\ne j}\,T_i(\gamma)/q_{0.5}(\chi^{2}_{m_i})=1$, with $T_i$ the energy statistic of view
$i$ and $m_i$ its degrees of freedom. A minority of faulty views can only raise $\hat\gamma$, which makes
the test conservative.

Two statistics are used. The energy $T_j=\mathbf{r}_j^{\top}M\mathbf{r}_j$, with $M$ the precision
$\Sigma_r^{-1}$ projected orthogonally to the alignment, has $m_j$ degrees of freedom based on
$n_{\mathrm{eff}}=\mathrm{tr}(C)^{2}/\mathrm{tr}(C^{2})$. With $m_j$ of order $10^{3}$ it is as much a test
of the noise level as of faults: a clean view whose noise is 2\% above the scene temperature already has a
23\% false-alarm rate at a nominal 5\%, so its threshold absorbs the per-view heterogeneity $\omega$
measured on development views. The region scan $\max_{\mathcal{R}}|z_{\mathcal{R}}|$ over appearance
segments $\mathcal{R}$ of view $j$,
$z_{\mathcal{R}}=\mathbf{g}_{\mathcal{R}}^{\top}M\mathbf{r}_j/\sqrt{\mathbf{g}_{\mathcal{R}}^{\top}M\mathbf{g}_{\mathcal{R}}}$,
with a Bonferroni threshold, is barely affected by $\omega$. For a region bias $\delta\,\mathbf{g}_{\mathcal{R}}$
it is $\mathcal{N}(\delta\kappa,1)$ with $\kappa^{2}=\mathbf{g}_{\mathcal{R}}^{\top}M\mathbf{g}_{\mathcal{R}}$, so
the minimum detectable bias at missed-detection probability $\beta_{\mathrm{md}}$ is
$(z_\alpha+z_{\beta_{\mathrm{md}}})/\kappa$, as in Baarda's reliability theory \cite{stat_teunissen_dia}.
Thresholds are set for a 5\% per-view false alarm and calibrated on clean development views (H5).

\subsection{The Outer Loop}
\label{sec:method:loop}
A single pass leaves the geometry shaped by every view, including the faulty ones. We therefore repeat
training, freezing, fusing and testing in rounds (Algorithm~\ref{alg:gate}), each warm-started from the
previous one with densification and pruning back on, and drop flagged views between rounds. A fixed
$\alpha$ per round would let the false-alarm rate grow with the number of rounds, so a Holm step-down
schedule runs over a family fixed in advance, $M=NR_{\max}$ tests: the $(k{+}1)$-th rejection overall uses
the threshold $\alpha/(M-k)$. Because the p-values change as the surface is retrained, the level of the
whole loop is calibrated by clean-run permutation. By default a round drops only the most significant
rejected view, with no readmission and $R_{\max}=10$. This guards against \emph{swamping}, in which a clean
view looks faulty only because a faulty view still shapes its cavity, and retraining also resolves
\emph{masking} among a minority of faulty views. The loop is iterative data snooping
\cite{stat_teunissen_dia} and hard EM \cite{stat_dempster_em}. A test costs $B$ backward passes, one render
and one $9\times9$ solve per touched surfel, so it is linear in views and pixels; the depth maps are cached.

\begin{algorithm}[t]
\caption{Train, freeze, fuse, test (outer loop)}
\label{alg:gate}
\begin{algorithmic}[1]
\REQUIRE cached depth maps $\{\hat d_i\}$; posed images; strip and time stamps; block size $B$; offline noise shape $\sigma(\cdot),\ell,\nu_{\mathrm{nug}}$; $\alpha$; $R_{\max}$
\STATE active views $\mathcal{A}\gets$ all $N$ views; rejections $k\gets0$; family $M\gets NR_{\max}$
\FOR{round $r=1,\dots,R_{\max}$}
  \STATE \textbf{train:} on $\mathcal{A}$, warm-started; $\boldsymbol\mu,\mathbf{R},\mathbf{s},\alpha$ on depth (opacity unfrozen); $(a_i,b_i)$ closed form; $\beta$ by least squares; densify only with $\ge2$ supporting views; prune
  \STATE \textbf{freeze} opacity; stop densification and pruning; fit $\boldsymbol\mu,\mathbf{R},\mathbf{s}$ and the spline control points (render-and-compare) to convergence; colour on the frozen geometry
  \STATE \textbf{fuse:} accumulate $\Lambda_k$ from the views in $\mathcal{A}$ at the MAP \hfill \eqref{eq:info}
  \FOR{each view $j\in\mathcal{A}$ \textbf{(test)}}
    \STATE block $\mathcal{B}_j$ of $B$ neighbouring frames; backward passes: $\sum_{i\in\mathcal{B}_j}\Lambda^{(i)}$, $\sum_{i\in\mathcal{B}_j}\nabla\mathcal{L}_i$
    \STATE cavity by block subtraction and Newton step; pose of $j$ without its depth, $\Sigma_q$ from the spline \hfill \eqref{eq:block}
    \STATE drop surfels by the support rules; render $\hat D_{-j},V_{-j}$ \hfill \eqref{eq:var}
    \STATE $\hat\gamma_{-j}$; mark untestable pixels; align; whiten; p-value $p_j$ \hfill \eqref{eq:res}
  \ENDFOR
  \STATE Holm step-down on the sorted $p_j$ with thresholds $\alpha/(M-k),\alpha/(M-k-1),\dots$
  \IF{no view is rejected}
    \STATE \textbf{stop}
  \ENDIF
  \STATE drop the most significant rejected view from $\mathcal{A}$; $k\gets k+1$
  \IF{more than $N/2$ views are excluded}
    \STATE scene-level alarm; \textbf{stop}
  \ENDIF
\ENDFOR
\STATE alarm if $R_{\max}$ was reached with a rejection; the last fit on $\mathcal{A}$ is the output
\end{algorithmic}
\end{algorithm}
