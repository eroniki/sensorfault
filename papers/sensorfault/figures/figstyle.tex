% Shared PGFPlots style for the sensorfault figures, after the data plots of Ambarkutuk et al.,
% Sensors 2024, 24(23):7744 (figures 5-8) and the figure conventions described in Sensors 2023,
% 23(23):9309: boxed axes with outward ticks and no grid, thin lines, the method in red and the baseline
% in black, a framed legend in a corner, sans-serif labels with units in square brackets, viridis for
% scalar maps, red x / black + markers, orange and maroon accents.
% Requires: pgfplots (compat >= 1.16), groupplots library, sansmath (sans-serif numerals with true minus signs).
\definecolor{sfred}{RGB}{220,20,20}
\definecolor{sfblack}{RGB}{0,0,0}
\definecolor{sforange}{RGB}{230,126,34}
\definecolor{sfmaroon}{RGB}{128,0,64}
\definecolor{sfgrey}{RGB}{150,150,150}
\definecolor{sflightgrey}{RGB}{225,225,225}
\pgfplotsset{
  colormap={sfviridis}{rgb255=(68,1,84) rgb255=(59,82,139) rgb255=(33,145,140) rgb255=(94,201,98) rgb255=(253,231,37)},
  colormap={sfrdbu}{rgb255=(33,102,172) rgb255=(146,197,222) rgb255=(247,247,247) rgb255=(244,165,130) rgb255=(178,24,43)},
  sfaxis/.style={
    font=\sffamily\scriptsize,
    tick label style={font=\sffamily\sansmath\scriptsize},
    every node near coord/.append style={font=\sffamily\sansmath\tiny},
    label style={font=\sffamily\scriptsize},
    title style={font=\sffamily\scriptsize},
    legend style={font=\sffamily\scriptsize, draw=sfgrey, rounded corners=1.5pt, fill=white, fill opacity=0.9, text opacity=1,
      inner sep=1.5pt, row sep=-1pt, legend cell align=left},
    tick align=outside, tick pos=left, major tick length=2.5pt,
    axis line style={line width=0.5pt}, every tick/.style={line width=0.5pt, black},
    grid=none, line width=0.8pt,
    every axis plot/.append style={line width=0.9pt},
    colormap name=sfviridis,
  },
  sfbar/.style={ybar, /pgf/bar width=5pt, every axis plot/.append style={line width=0.4pt},
    legend image code/.code={\draw[##1, line width=0.4pt] (0cm,-0.07cm) rectangle (0.22cm,0.09cm);}},
  sfbarred/.style={fill=sfred, draw=sfred!60!black},
  sfbarblack/.style={fill=black!85, draw=black},
  sfbargrey/.style={fill=sfgrey, draw=black!70},
  sfimage/.style={sfaxis, enlargelimits=false, axis on top, xtick=\empty, ytick=\empty,
    axis line style={line width=0.4pt}, scale only axis},
  sfcolorbar/.style={sfaxis, hide axis, scale only axis, height=0pt, width=#1, colorbar horizontal,
    colorbar style={height=0.13cm, width=#1, xtick pos=left, tick align=outside, major tick length=1.5pt,
      xticklabel style={font=\sffamily\sansmath\tiny, /pgf/number format/fixed, /pgf/number format/precision=2},
      xlabel style={font=\sffamily\tiny, yshift=2pt}, every axis plot/.append style={}}},
}
