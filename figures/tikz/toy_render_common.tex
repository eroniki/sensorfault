% Toy residual maps for one fault family (\toyfault = square|warp|edge): clean versus faulted view, true and estimated covariance.
% Data: data/toy_<fault>_<variant>_<clean|fault>_<r|e>.csv, data/toy_render_macros.tex (T, threshold, shared colour limits), data/toy_outline_<fault>.csv.
\input{\tikzdir/data/toy_render_macros.tex}
\newcommand{\toyg}[1]{\csname #1\endcsname}
% #1 variant, #2 state, #3 kind (r|e), #4 colormap, #5 extra axis options
\newcommand{\toypanel}[6]{%
  \edef\vlim{\toyg{ToyV#3@\toyfault @#1}}%
  \addplot[matrix plot*, mesh/cols=16, mesh/rows=16, point meta=explicit, point meta min=-\vlim, point meta max=\vlim, colormap name=#4]
    table[x=x, y=y, meta=z, col sep=space]{\tikzdir/data/toy_\toyfault _#1_#2_#3.csv};
  \ifnum#6=1 \addplot[black, line width=0.6pt, unbounded coords=jump, forget plot] table[x=x, y=y]{\tikzdir/data/toy_outline_\toyfault.csv};\fi}
\pgfplotsset{toyax/.style={axis on top, enlargelimits=false, xmin=0, xmax=16, ymin=0, ymax=16, xtick=\empty, ytick=\empty, width=0.135\figw, height=0.135\figw,
  colorbar style={width=0.18cm, tick label style={font=\figfont\tiny}, tick align=outside, tick style={line width=0.4pt}}}}
