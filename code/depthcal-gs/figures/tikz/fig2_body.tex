\newcommand{\figtwocell}[4]{% #1 row index, #2 column key, #3 x, #4 y   ; leaky row (2): passes of poisoned/invariant must be 0 (caught)
  \pgfplotstablegetelem{#1}{n}\of\figtwomat \edef\nn{\pgfplotsretval}
  \pgfplotstablegetelem{#1}{#2}\of\figtwomat \edef\kk{\pgfplotsretval}
  \ifnum#1=2
    \ifx#2\relax\fi
    \edef\isleaky{1}
  \else \edef\isleaky{0}\fi
  \def\cellok{0}
  \ifnum\isleaky=1
    \IfSubStr{#2}{sensitive}{\ifnum\kk=\nn\def\cellok{1}\fi}{\ifnum\kk=0\def\cellok{1}\fi}
  \else
    \ifnum\kk=\nn\def\cellok{1}\fi
  \fi
  \ifnum\cellok=1 \def\cellcol{ambPass}\else\def\cellcol{ambFail}\fi
  \fill[\cellcol] (#3-0.46,#4-0.4) rectangle (#3+0.46,#4+0.4);
  \node[font=\figfont\footnotesize] at (#3,#4) {%
    \ifnum\isleaky=1 \IfSubStr{#2}{sensitive}{\kk/\nn\ pass}{\number\numexpr\nn-\kk\relax/\nn\ caught}\else \kk/\nn\ pass\fi};}
\begin{tikzpicture}
