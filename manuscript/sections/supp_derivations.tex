\section{Derivations}
\label{sup:derivations}

\subsection{Ray-Surfel Depth and Its Jacobian}
Let the camera have centre $\mathbf{c}$ and ray $\mathbf{w}=\mathbf{R}_c\mathbf{K}^{-1}[u\ v\ 1]^{\top}$
(so the parameter along $\mathbf{w}$ is the z-depth). The surfel plane is
$\mathbf{n}^{\top}(\mathbf{x}-\boldsymbol\mu)=0$, so $d=\mathbf{n}^{\top}(\boldsymbol\mu-\mathbf{c})/\mathbf{n}^{\top}\mathbf{w}$
and $\mathbf{x}=\mathbf{c}+d\mathbf{w}$. Differentiating
$\mathbf{n}^{\top}(\mathbf{c}+d\mathbf{w}-\boldsymbol\mu)=0$ gives
$\delta\mathbf{n}^{\top}(\mathbf{x}-\boldsymbol\mu)+\mathbf{n}^{\top}\mathbf{w}\,\delta d-\mathbf{n}^{\top}\delta\boldsymbol\mu=0$, so
\[
\delta d=\frac{\mathbf{n}^{\top}\delta\boldsymbol\mu-\delta\mathbf{n}^{\top}(\mathbf{x}-\boldsymbol\mu)}{\mathbf{n}^{\top}\mathbf{w}} .
\]
With $\mathbf{R}\leftarrow\mathbf{R}\exp([\delta\boldsymbol\phi]_\times)$ and $\mathbf{R}=[\mathbf{t}_1\ \mathbf{t}_2\ \mathbf{n}]$,
$\delta\mathbf{n}=\mathbf{R}(\delta\boldsymbol\phi\times\mathbf{e}_3)=\delta\phi_2\mathbf{t}_1-\delta\phi_1\mathbf{t}_2$, which has no normal
component. Hence $\delta d$ depends on $\boldsymbol\theta$ only through
$(\nu,\tau_1,\tau_2)=(\mathbf{n}^{\top}\delta\boldsymbol\mu,\ \delta\phi_2,\ -\delta\phi_1)$, which gives \eqref{M-eq:jac}.
The in-plane components of $\delta\boldsymbol\mu$, the spin $\delta\phi_3$ and the scales do not change
the ray-surfel depth to first order and keep their prior information $\Lambda_0$; they must not be
reported as learned surface uncertainty.

\runin{Opacity.} With $w_k=\alpha_kg_kT_k$, $T_k=\prod_{l\prec k}(1-\alpha_lg_l)$, $A=\sum_kw_kd_k$,
$W=\sum_kw_k$ and $\hat D=A/W$,
\[
\frac{\partial w_m}{\partial\alpha_k}=\begin{cases}g_kT_k,&m=k,\\ -\dfrac{g_k\,w_m}{1-\alpha_kg_k},&m\succ k,\\ 0,&m\prec k,\end{cases}
\]
\[
\frac{\partial\hat D}{\partial\,\mathrm{logit}\,\alpha_k}=\alpha_k(1-\alpha_k)\,\frac{\partial_{\alpha_k}A-\hat D\,\partial_{\alpha_k}W}{W},
\]
which includes the transmittance of the surfels behind $k$. The code checks this and \eqref{M-eq:jac}
against finite differences of the exact compositing and ray-surfel depth.

\subsection{Information Update, Posterior and Newton Leave-Out}
Linearise the measurement of view $i$ at the MAP $\hat{\boldsymbol\theta}$: $\hat d_i(u)=a_i\bigl(\hat D_i(u)+\sum_k
\mathbf{g}_{uk}\boldsymbol\theta_k\bigr)+b_i+\varepsilon_i(u)$. With independent pixel noise and independent
surfel blocks the Gaussian likelihood of view $i$ contributes \eqref{M-eq:info}, and with opacity frozen the
posterior of the geometric coordinates is Gaussian with the precision of the geometric block. For the
linear model, $\Lambda\hat{\boldsymbol\theta}=\boldsymbol\eta$ and
$\Lambda^{(-j)}\hat{\boldsymbol\theta}^{(-j)}=\boldsymbol\eta-\boldsymbol\eta^{(j)}$, so
\[
\hat{\boldsymbol\theta}^{(-j)}-\hat{\boldsymbol\theta}=\bigl(\Lambda^{(-j)}\bigr)^{-1}\bigl(\Lambda^{(j)}\hat{\boldsymbol\theta}-\boldsymbol\eta^{(j)}\bigr)
=\bigl(\Lambda^{(-j)}\bigr)^{-1}\nabla\mathcal{L}_j(\hat{\boldsymbol\theta}),
\]
which is \eqref{M-eq:loo}: one Newton step on the loss without view $j$, started at the full MAP (the
infinitesimal jackknife). It is exact for the linear-Gaussian model and a first-order approximation
otherwise; the code checks that it equals a fusion recomputed without the view. The shared bias $\beta$
has scalar normal equations $\sum_i\lambda^{\beta}_i\,\beta=\sum_i\eta^{\beta}_i$, with $\lambda^{\beta}_i=\sum_u a_i^{2}d_i(u)^{4}/\sigma_i^{2}(u)$, so its leave-view-out value subtracts
$(\lambda^{\beta}_j,\eta^{\beta}_j)$.

\subsection{Solution Separation Equals the Leave-One-Out Innovation}
\begin{proposition}
\label{prop:loo}
Let view $j$ measure a state $\mathbf{x}$ directly, $\mathbf{z}_j=\mathbf{x}+\mathbf{v}_j$ with noise covariance
$\Sigma_j$, let $\hat{\mathbf{x}}_{-j}$ be the estimate without it, with covariance $P_{-j}$, and let
$\mathbf{q}=\mathbf{z}_j-\hat{\mathbf{x}}_{-j}$, $S=\Sigma_j+P_{-j}$. Then the solution-separation statistic
$\mathbf{d}^{\top}\mathrm{Cov}(\mathbf{d})^{-1}\mathbf{d}$, $\mathbf{d}=\hat{\mathbf{x}}_{\mathrm{all}}-\hat{\mathbf{x}}_{-j}$, equals the
leave-one-out statistic $\mathbf{q}^{\top}S^{-1}\mathbf{q}$.
\end{proposition}
\emph{Proof.} $\mathbf{d}=P_{-j}S^{-1}\mathbf{q}$ and $\mathrm{Cov}(\mathbf{d})=P_{-j}-P_{\mathrm{all}}=P_{-j}S^{-1}P_{-j}$, so
\[
\mathbf{d}^{\top}\mathrm{Cov}(\mathbf{d})^{-1}\mathbf{d}
=\mathbf{q}^{\top}S^{-1}P_{-j}\bigl(P_{-j}S^{-1}P_{-j}\bigr)^{-1}P_{-j}S^{-1}\mathbf{q}
=\mathbf{q}^{\top}S^{-1}\mathbf{q}.\qquad\square
\]
For the normal offset of one surfel, with information $\lambda_j$ from view $j$ and $\lambda_{-j}$ from the
others, both statistics equal $(\nu_j-\nu_{-j})^{2}\lambda_j\lambda_{-j}/(\lambda_j+\lambda_{-j})$
\cite{flt_raim_joerger}.

\subsection{Rendered Variance and the Whitened Residual}
With independent blocks and opacity frozen,
\[
\mathrm{Var}\,\hat D_{-j}(u)=\sum_kw_{uk}^{2}J_{uk}P_k^{(-j)}J_{uk}^{\top},
\]
with $P_k^{(-j)}$ the inverse of $\Lambda_k^{(-j)}$ on the nine geometric coordinates, which is \eqref{M-eq:var}. The residual \eqref{M-eq:res} is the sum of the depth noise of view $j$ and the
error of the leave-$j$-out surface, which are independent because view $j$ was subtracted, so its
covariance is $\gamma^{2}DCD+a_j^{2}UP^{(-j)}U^{\top}+J_q\Sigma_qJ_q^{\top}$, the last term being the error of view
$j$'s pose, which was estimated without its depth. Pixels that one surfel renders are correlated through
that surfel's error, which $UP^{(-j)}U^{\top}$ keeps; cross-surfel correlations are dropped. With
$W=(\gamma^{2}DCD)^{-1}$ (diagonal when $C=I$) and $\Lambda^{(-j)}=(P^{(-j)})^{-1}$, the Woodbury identity
gives
\[
\bigl(W^{-1}+UP^{(-j)}U^{\top}\bigr)^{-1}=W-WU\bigl(\Lambda^{(-j)}+U^{\top}WU\bigr)^{-1}U^{\top}W,
\]
and $\Lambda^{(-j)}+U^{\top}WU$ is block-diagonal with one $3\times3$ block per surfel, so applying the
precision costs one pass over the pixels. The code checks it against a dense inverse. When each
pixel is rendered by one surfel the blocks are exact and the whitened residual is $\mathcal{N}(0,1)$;
when footprints overlap, the dropped cross-surfel terms make it overconfident. The code reproduces both
cases on a synthetic plane, and also the overconfidence caused by spatially correlated noise, which
deflating each view's information reduces.

\subsection{Minimum Detectable Bias of a Region}
For a bias $\delta$ on a region $\mathcal{R}$ with pattern $\mathbf{g}_{\mathcal{R}}$, the matched statistic is
$\mathcal{N}(\delta\kappa,1)$ with $\kappa^{2}=\mathbf{g}_{\mathcal{R}}^{\top}M\mathbf{g}_{\mathcal{R}}$ and $M$ the precision projected
orthogonally to the alignment $A=[\hat{\mathbf{D}}_{-j},\mathbf{1}]$,
$M=\Sigma_r^{-1}-\Sigma_r^{-1}A(A^{\top}\Sigma_r^{-1}A)^{-1}A^{\top}\Sigma_r^{-1}$; the minimum detectable bias at
false-alarm $\alpha$ and missed-detection $\beta_{\mathrm{md}}$ is
$(z_{\alpha}+z_{\beta_{\mathrm{md}}})/\kappa$, and a scan over $N_{\mathcal{R}}$ regions replaces $z_\alpha$ by
$z_{\alpha/N_{\mathcal{R}}}$. A pattern in the span of $A$ (a whole-view affine error) has $\kappa=0$ and is not
detectable, which is why it is not a fault. With spatially correlated noise, $\kappa$ follows from
$\Sigma_r$, not from the pixel count, and the effective number of pixels
$n_{\mathrm{eff}}=\mathrm{tr}(C)^{2}/\mathrm{tr}(C^{2})$ summarises the energy statistic.

\subsection{Scene Temperature and Per-View Heterogeneity}
With $\gamma^{2}DCD$ dominating $\Sigma_r$, $T_i(\gamma)=T_i(1)/\gamma^{2}$, so
\[
\hat\gamma^{2}_{-j}=\operatorname*{median}_{i\ne j}\,\frac{T_i(1)}{q_{0.5}(\chi^{2}_{m_i})};
\]
in general it is a monotone one-dimensional root. Its relative standard deviation is about $1.25\sqrt{2/n_{\mathrm{eff}}}/\sqrt{N-1}$
(0.8\% at $n_{\mathrm{eff}}=10^{3}$, $N=30$). A fraction $\epsilon<\tfrac12$ of faulty views moves the
median to the $1/(2(1-\epsilon))$ quantile of the clean distribution, raising $\hat\gamma$: the test
becomes conservative, with breakdown at $\epsilon=\tfrac12$, consistent with the minority assumption.
As in temperature scaling \cite{cal_guo_temperature}, one scale does not change the ranking of views
within a scene, only the operating point; unlike there, $\gamma$ is fitted on the unlabelled views of
the test scene, which is valid only leave-view-out and under the minority assumption.

The energy statistic has $m$ of order $n_{\mathrm{eff}}$ degrees of freedom, so it is sensitive to the
noise level of a view. If view $j$'s true noise scale is $r$ times the temperature, its false-alarm rate
at nominal $\alpha=0.05$ is $P(r^{2}\chi^{2}_m>q_{0.95}(\chi^{2}_m))$, which is 0.050, 0.117, 0.229, 0.710
and 0.995 at $r=1$, 1.01, 1.02, 1.05 and 1.10 for $m=10^{3}$. If $\log r\sim\mathcal{N}(0,\omega^{2})$
across clean views, holding the average false-alarm rate at $\alpha$ needs the threshold
$c/m=1.075$, 1.102, 1.196 and 1.400 for $\omega=0$, 0.02, 0.05 and 0.10, and the raised threshold raises
the energy minimum detectable bias. The matched region scan only scales $z$ by $1/r$, so with
$N_{\mathcal{R}}=100$ regions its family false-alarm rate stays below 0.16 up to $r=1.10$. Hence $\omega$ is
measured first (H1), the region scan leads if $\omega\gtrsim0.05$, and the energy threshold uses
$n_{\mathrm{eff}}$, not the pixel count. The code reproduces the false-alarm rates and thresholds above.

\subsection{Holm Step-Down Across Rounds}
The loop may run at most $M=NR_{\max}$ view tests: every view in every round, including re-tests of a
view that was tested before and, in the readmission ablation, tests of dropped views. Fixing $M$ before
the loop starts and rejecting the $(k{+}1)$-th view overall only when its p-value is at most
$\alpha/(M-k)$ gives, by the union bound over the at most $M$ tests of clean views, a family-wise
false-alarm rate of at most $\alpha$ for the Bonferroni part of the schedule. The step-down relaxation
from $\alpha/M$ to $\alpha/(M-k)$ is Holm's argument, which assumes a fixed set of p-values; between rounds
the surface is retrained and the p-values change, so the overall level is checked by clean-run
permutation of the whole loop. With $N=30$ and $R_{\max}=10$, $M=300$, and the first rejection needs
$p\le1.7\times10^{-4}$ at $\alpha=0.05$. The code checks the counting, the step-down within a round,
masking (a faulty view revealed only after another is dropped), swamping (a clean view flagged only while a
faulty view is active, excluded by the all-rejected policy but not by the default), readmission, and a
family-wise false-alarm rate of at most $\alpha$ under a null simulation.

\subsection{Leave-Block-Out}
For a block $\mathcal{B}$ of views, the linear model gives $\Lambda\hat{\boldsymbol\theta}=\boldsymbol\eta$ and
$\Lambda^{(-\mathcal{B})}\hat{\boldsymbol\theta}^{(-\mathcal{B})}=\boldsymbol\eta-\sum_{i\in\mathcal{B}}\boldsymbol\eta^{(i)}$, with
$\Lambda^{(-\mathcal{B})}=\Lambda-\sum_{i\in\mathcal{B}}\Lambda^{(i)}$, so
\[
\hat{\boldsymbol\theta}^{(-\mathcal{B})}-\hat{\boldsymbol\theta}=\bigl(\Lambda^{(-\mathcal{B})}\bigr)^{-1}\sum_{i\in\mathcal{B}}\nabla\mathcal{L}_i(\hat{\boldsymbol\theta}),
\]
which is \eqref{M-eq:block}. It equals removing the views of the block one after another, since messages add,
and it equals a fusion recomputed without the block; the code checks both, and that the cavity of a block
carries less normal-offset information than the cavity of one of its views. Only the tested view's
residual enters its test, so the Holm family counts views, not blocks.

\subsection{Trajectory Spline and Pose Uncertainty}
With uniform knots $t_0+i\Delta t$ and $s=(t-t_0)/\Delta t$, $i=\lfloor s\rfloor$, $u=s-i$, the position is
$\mathbf{p}(t)=\sum_{m=0}^{3}b_m(u)\,\mathbf{c}_{i+m}$ with the uniform cubic B-spline weights
$[b_0,\dots,b_3]=[1,u,u^{2},u^{3}]\,\mathsf{M}/6$, which sum to one and reproduce straight, constant-speed
strips exactly. Rotations use the cumulative form
$\mathbf{R}(t)=\mathbf{R}_i\prod_{m=1}^{3}\exp\bigl(\tilde b_m(u)\log(\mathbf{R}_{i+m-1}^{\top}\mathbf{R}_{i+m})\bigr)$ with
$\tilde b_m=\sum_{l\ge m}b_l$ \cite{traj_sommer_bspline}; a constant-rate turn is reproduced exactly. The
control points are fitted by weighted least squares to the GNSS/IMU positions with a second-difference
prior, which gives the control-point information $\Lambda_{\mathrm{ctrl}}$; render-and-compare terms add to
it. For the tested view, its own term is subtracted, and $\Sigma_q$ at its time stamp is
$\mathbf{b}(t_j)\Lambda_{\mathrm{ctrl}}^{-1}\mathbf{b}(t_j)^{\top}$ per axis for the position, with $\mathbf{b}(t_j)$ the basis row. The
code checks the partition of unity and linear precision, the error reduction on a noisy simulated
two-strip flight, the position variance against Monte Carlo, the subtraction of the tested view's term,
and the rotation spline.
